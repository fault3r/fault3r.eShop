        private static Mock<IMongoCollection<ItemDocument>> CreateCollectionMock()
        {
            return new Mock<IMongoCollection<ItemDocument>>();
        }

        private static Mock<IFindFluent<ItemDocument, ItemDocument>> CreateFindFluentMock()
        {
            var mock = new Mock<IFindFluent<ItemDocument, ItemDocument>>();
            mock.Setup(m => m.ToListAsync(It.IsAny<CancellationToken>()))
                .ReturnsAsync(new List<ItemDocument>());
            mock.Setup(m => m.FirstOrDefaultAsync(It.IsAny<CancellationToken>()))
                .Returns(Task.FromResult<ItemDocument>(null!));
            return mock;
        }

        [Fact]
        public async Task GetAllAsync_ReturnsOkWithItems()
        {
            var document = new ItemDocument { Name = "name", Description = "d", Price = 1m, Pictures = new[] { "p" } };
            var list = new List<ItemDocument> { document };

            var findFluent = new Mock<IFindFluent<ItemDocument, ItemDocument>>();
            findFluent.Setup(f => f.ToListAsync(It.IsAny<CancellationToken>())).Returns(Task.FromResult(list));

            var collection = CreateCollectionMock();
            collection.Setup(c => c.Find(It.IsAny<FilterDefinition<ItemDocument>>(), It.IsAny<FindOptions>()))
                .Returns(findFluent.Object);

            var context = new Mock<CatalogContext>();
            context.Setup(c => c.Documents).Returns(collection.Object);

            var repo = new CatalogRepository(context.Object);

            var result = await repo.GetAllAsync();

            Assert.Equal(200, result.Code);
            Assert.NotNull(result.Items);
            Assert.Equal(1, result.Items.Count());
            Assert.Equal(document.Name, result.Items.First().Name);
        }

        [Fact]

        [Fact]
        public async Task DeleteAsync_NotFound()
        {
            var collection = CreateCollectionMock();
            // Create a DeleteResult with DeletedCount = 0 via reflection to avoid API differences
            var acknowledgedMethod = typeof(DeleteResult).GetMethod("Acknowledged", System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.Static);
            var noneResult = acknowledgedMethod != null ? (DeleteResult)acknowledgedMethod.Invoke(null, new object[] { 0L }) : (DeleteResult)Activator.CreateInstance(typeof(DeleteResult), true)!;
            collection.Setup(c => c.DeleteOneAsync(It.IsAny<FilterDefinition<ItemDocument>>(), It.IsAny<CancellationToken>()))
                .ReturnsAsync(noneResult);

            var context = new Mock<CatalogContext>();
            context.Setup(c => c.Documents).Returns(collection.Object);

            var repo = new CatalogRepository(context.Object);

            var result = await repo.DeleteAsync(ObjectId.GenerateNewId().ToString());

            Assert.Equal(404, result.Code);
        }

        [Fact]
        public async Task DeleteAsync_Success()
        {
            var collection = CreateCollectionMock();
            var acknowledgedMethod = typeof(DeleteResult).GetMethod("Acknowledged", System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.Static);
            var oneResult = acknowledgedMethod != null ? (DeleteResult)acknowledgedMethod.Invoke(null, new object[] { 1L }) : (DeleteResult)Activator.CreateInstance(typeof(DeleteResult), true)!;
            collection.Setup(c => c.DeleteOneAsync(It.IsAny<FilterDefinition<ItemDocument>>(), It.IsAny<CancellationToken>()))
                .ReturnsAsync(oneResult);

            var context = new Mock<CatalogContext>();
            context.Setup(c => c.Documents).Returns(collection.Object);

            var repo = new CatalogRepository(context.Object);

            var result = await repo.DeleteAsync(ObjectId.GenerateNewId().ToString());

            Assert.Equal(204, result.Code);
        }
