using System;
using System.Threading;
using System.Threading.Tasks;
using AccountService.Infrastructure.Persistence;
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Storage;

namespace AccountService.Infrastructure.Persistence
{
    public class EfUnitOfWork : IUnitOfWork, IDisposable
    {
        private readonly AccountDbContext _db;
        private bool _disposed;

        public EfUnitOfWork(AccountDbContext db) => _db = db ?? throw new ArgumentNullException(nameof(db));

        /// <summary>
        /// Saves changes on the DbContext. If there is an ambient transaction opened by caller
        /// (BeginTransaction / ExecuteTransactionalAsync), DbContext will enlist in it. This method
        /// does not start or commit transactions by itself.
        /// </summary>
        public async Task<int> SaveChangesAsync(CancellationToken ct = default)
        {
            // Let DbContext handle concurrency/validation exceptions; bubble them up to caller.
            return await _db.SaveChangesAsync(ct);
        }

        /// <summary>
        /// Execute a delegate inside a database transaction. Ensures action's changes and any
        /// Outbox rows are committed together. Commits on success and rolls back on exception.
        /// </summary>
        public async Task ExecuteTransactionalAsync(Func<Task> action, CancellationToken ct = default)
        {
            if (action == null) throw new ArgumentNullException(nameof(action));

            // If a transaction already exists (higher layer), reuse it.
            var currentTx = _db.Database.CurrentTransaction;
            if (currentTx != null)
            {
                // Already inside a transaction scope; just execute action
                await action();
                return;
            }

            // Otherwise start a new transaction with a reasonable isolation level.
            await using var tx = await _db.Database.BeginTransactionAsync(System.Data.IsolationLevel.ReadCommitted, ct);
            try
            {
                await action();

                // Persist changes made by action (including Outbox rows)
                await _db.SaveChangesAsync(ct);

                await tx.CommitAsync(ct);
            }
            catch
            {
                try { await tx.RollbackAsync(ct); } catch { /* swallow rollback failures */ }
                throw;
            }
        }

        public void Dispose()
        {
            if (_disposed) return;
            _db?.Dispose();
            _disposed = true;
            GC.SuppressFinalize(this);
        }
    }
}
